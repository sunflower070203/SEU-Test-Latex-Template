  \examquestion 设 $f(x)=x^{2}$，则 $f'(1)=\underline{\hspace{2cm}}$。

%% ======================= 二、计算题 =======================
%% 计算题每题后面留答题空间：方括号里写题间距（原卷约 9cm）
\examsection{计算下列各题（本题共2小题，每小题7分，满分14分）}
\begin{examquestions}[9cm]
  \examquestion 求函数 $z=x^{2}+y^{2}$ 在闭区域
    $D=\{(x,y)\mid x^{2}+y^{2}\le 1\}$ 上的最大值与最小值。
  \examquestion 计算曲线积分 $\displaystyle\oint_{L}(x+y)\,\mathrm{d}s$，
    其中 $L$ 为圆周 $x^{2}+y^{2}=2x$。
\end{examquestions}

%% ======================= 三、单个大题（无小题编号）=======================
\examsection{（本题满分8分）}
\begin{examproblem}
  已知曲线 $L$ 为闭曲线
  $\begin{cases}x^{2}+y^{2}+z^{2}=1,\\ z=x,\end{cases}$
  从 $z$ 轴正向负向看去为逆时针方向，求曲线积分
  $\displaystyle\oint_{L}(y+z)\,\mathrm{d}x+(z^{2}-x^{2}+y)\,\mathrm{d}y
   +x^{2}y^{2}\,\mathrm{d}z$。
\end{examproblem}
